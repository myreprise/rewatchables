<script>
  // GitHub Pages serves this page for any address that does not exist. Until
  // October 2026 every film, keyword and person had a page of its own at
  // /movie/heat-1995, /keyword/heist and so on; now one page shows each kind,
  // reading the address (movie?id=heat-1995). Forward an old link to its new
  // address, so links shared before the change keep working. A plain script,
  // not a Framework cell, so it runs even where this page's assets do not
  // resolve (it is served at the old, deeper path).
  (function () {
    var kinds = {movie: "movie", keyword: "keyword", person: "person",
                 director: "director", actor: "actor", upcoming: "request"};
    var m = location.pathname.match(/^(.*?)\/(movie|keyword|person|director|actor|upcoming)\/([a-z0-9-]+)\/?$/);
    if (m) location.replace(m[1] + "/" + kinds[m[2]] + "?id=" + m[3]);
  })();
</script>
